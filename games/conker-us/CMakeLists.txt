set(RECOMP_ROM "${CMAKE_SOURCE_DIR}/local/conker.us.z64")
set(RECOMP_SAVE "${CMAKE_SOURCE_DIR}/local/saves/conker-us.eeprom")
set(RECOMP_GAME_GLES2_SUPPORTED ON)
if(RECOMP_RENDERER STREQUAL "rt64")
    set(RECOMP_GAME_SOURCES graphics.cpp)
endif()
include("${CMAKE_SOURCE_DIR}/cmake/Game.cmake")
if(EMSCRIPTEN)
    target_compile_options(${RECOMP_GAME} PRIVATE --use-port=zlib)
    target_link_options(${RECOMP_GAME} PRIVATE --use-port=zlib)
else()
    find_package(ZLIB REQUIRED)
    target_link_libraries(${RECOMP_GAME} PRIVATE ZLIB::ZLIB)
endif()
if(RECOMP_RENDERER STREQUAL "rt64")
    target_include_directories(${RECOMP_GAME} SYSTEM PRIVATE ${RECOMP_RT64_INCLUDES})
    target_compile_definitions(${RECOMP_GAME} PRIVATE HLSL_CPU FFX_GCC IMGUI_IMPL_VULKAN_NO_PROTOTYPES)
endif()
