    add_executable(sm64_native_full_rt64 app.c "${RECOMP_FULL_GENERATED}/functions.c")
    set_target_properties(sm64_native_full_rt64 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

    target_compile_definitions(sm64_native_full_rt64 PRIVATE RECOMP_WITH_RT64)

        target_compile_definitions(sm64_native_full_rt64 PRIVATE RECOMP_WITH_AUDIO)
        target_link_libraries(sm64_native_full_rt64 PRIVATE recomp_audio)
    endif()
