cmake_minimum_required(VERSION 3.20)
# Select defaults before project() enables C; preserve explicit compiler/toolchain choices.
if(NOT CMAKE_HOST_WIN32 AND NOT CMAKE_TOOLCHAIN_FILE AND "$ENV{CMAKE_TOOLCHAIN_FILE}" STREQUAL ""
        AND NOT DEFINED CMAKE_C_COMPILER AND "$ENV{CC}" STREQUAL "")
    find_program(CMAKE_C_COMPILER NAMES clang gcc REQUIRED)
endif()
project(v64recomp LANGUAGES C)
include(cmake/SourceGroups.cmake)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# Static runtime/backend libraries must also support relocation into PIE games.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
configure_file(LICENSE "${CMAKE_BINARY_DIR}/bin/LICENSE" COPYONLY)
configure_file(THIRD_PARTY_NOTICES.md "${CMAKE_BINARY_DIR}/bin/THIRD_PARTY_NOTICES.md" COPYONLY)
configure_file(licenses/Mupen64Plus-rsp-hle.txt "${CMAKE_BINARY_DIR}/bin/licenses/Mupen64Plus-rsp-hle.txt" COPYONLY)
configure_file(licenses/minicoro.txt "${CMAKE_BINARY_DIR}/bin/licenses/minicoro.txt" COPYONLY)

# One native game per build tree. Use a separate directory for each mode.
set(RECOMP_GAME "sm64-us" CACHE STRING "Game to build, or none for shared libraries")
set(RECOMP_GAMES sm64-us goldeneye-us shadowman-us quake-us duke-us nightmare-us oot-us-v12 perfectdark-us-v11 conker-us duke-zero-hour-us hexen-us rayman2-us quake2-us re2-us-v1 turok-us-v12 1080-us banjo-kazooie-us twine-us)
list(APPEND RECOMP_GAMES bad-apple media-player)
set_property(CACHE RECOMP_GAME PROPERTY STRINGS none ${RECOMP_GAMES})
if(NOT RECOMP_GAME STREQUAL "none" AND NOT RECOMP_GAME IN_LIST RECOMP_GAMES)
    message(FATAL_ERROR "Unknown RECOMP_GAME=${RECOMP_GAME}. Choose: ${RECOMP_GAMES};none")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Release or Debug" FORCE)
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|Debug)$")
    message(FATAL_ERROR "CMAKE_BUILD_TYPE must be Release or Debug")
endif()
# Each solution exposes the selected mode; both compile the same game main.c.
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "${CMAKE_BUILD_TYPE}" CACHE STRING "Selected game mode" FORCE)
endif()
include(cmake/Release.cmake)
if(MSVC)
    if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
        if(CMAKE_BUILD_TYPE STREQUAL "Debug")
            set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
        else()
            set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
        endif()
    endif()
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN SDL_MAIN_HANDLED)
    add_compile_options(/utf-8 /Zi /FS /Zc:preprocessor "$<$<COMPILE_LANGUAGE:C>:/experimental:c11atomics>")
endif()
set(RECOMP_RUN_ARGS "" CACHE STRING "Extra executable arguments for the run target (semicolon separated)")
if(NOT RECOMP_UPX_EXECUTABLE)
    # Clear empty entries from older configurations so find_program searches.
    unset(RECOMP_UPX_EXECUTABLE CACHE)
    unset(RECOMP_UPX_EXECUTABLE)
    find_program(RECOMP_UPX_EXECUTABLE NAMES upx upx-ucl
        DOC "Path to UPX executable for the pack target")
endif()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(recomp_runtime STATIC runtime/host.c runtime/memory.c runtime/dispatch.c runtime/live_patch.c runtime/fpu.c runtime/scheduler.c
    runtime/os.c runtime/bootstrap.c runtime/time.c runtime/platform.c runtime/tasks.c runtime/io.c runtime/video.c runtime/profiler.c
    runtime/cheats.c runtime/timing.c runtime/fov_control.c)
target_link_libraries(recomp_runtime PUBLIC OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
    find_package(PThreads4W REQUIRED)
    target_link_libraries(recomp_runtime PUBLIC PThreads4W::PThreads4W dbghelp)
else()
    target_link_libraries(recomp_runtime PUBLIC m)
endif()
target_include_directories(recomp_runtime PUBLIC include)
target_include_directories(recomp_runtime PRIVATE third_party/minicoro)
target_sources(recomp_runtime PRIVATE include/recomp.h include/runtime.h include/host.h
    include/runtime_debug.h include/profiler.h include/cheats.h include/timing.h third_party/minicoro/minicoro.h)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(recomp_runtime PUBLIC RECOMP_DEBUG)
    target_sources(recomp_runtime PRIVATE runtime/debug.c)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_program(RECOMP_ADDR2LINE NAMES llvm-addr2line addr2line REQUIRED)
        target_compile_definitions(recomp_runtime PRIVATE RECOMP_ADDR2LINE="${RECOMP_ADDR2LINE}")
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(recomp_runtime PRIVATE -Wall -Wextra -Werror -fno-fast-math -frounding-math -ffp-contract=off)
    target_compile_options(recomp_runtime PRIVATE -g1 -fno-omit-frame-pointer)
endif()
if(MSVC)
    target_compile_options(recomp_runtime PRIVATE /W4 /fp:strict)
endif()
add_library(recomp_rsp_memory STATIC third_party/rsp-hle/memory.c)
target_sources(recomp_rsp_memory PRIVATE third_party/rsp-hle/memory.h
    third_party/rsp-hle/common.h third_party/rsp-hle/hle_internal.h)
if(NOT MSVC)
    target_compile_options(recomp_rsp_memory PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
endif()
add_library(recomp_jpeg STATIC backends/jpeg.c third_party/rsp-hle/jpeg.c)
target_sources(recomp_jpeg PRIVATE include/recomp_jpeg.h)
target_include_directories(recomp_jpeg PRIVATE third_party/rsp-hle)
target_link_libraries(recomp_jpeg PUBLIC recomp_runtime recomp_rsp_memory)
if(MSVC)
    target_compile_options(recomp_jpeg PRIVATE /W4 /fp:strict)
else()
    target_compile_options(recomp_jpeg PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing -fno-fast-math -ffp-contract=off)
endif()
set_property(SOURCE third_party/rsp-hle/jpeg.c APPEND PROPERTY COMPILE_DEFINITIONS
    rsp_break=recomp_jpeg_break HleWarnMessage=recomp_jpeg_warn HleVerboseMessage=recomp_jpeg_verbose)
find_package(SDL2 REQUIRED)
add_library(recomp_rsp STATIC runtime/rsp.c runtime/rsp_vector.c runtime/rsp_worker.c include/rsp.h include/rsp_worker.h)
target_link_libraries(recomp_rsp PUBLIC recomp_runtime)
if(MSVC)
    target_compile_options(recomp_rsp PRIVATE /W4)
else()
    target_compile_options(recomp_rsp PRIVATE -Wall -Wextra -Werror -g1 -fno-omit-frame-pointer)
endif()
add_library(recomp_rdp STATIC backends/rdp_triangle.c include/rdp_triangle.h)
target_link_libraries(recomp_rdp PUBLIC recomp_runtime)
if(MSVC)
    target_compile_options(recomp_rdp PRIVATE /W4 /fp:strict)
else()
    target_compile_options(recomp_rdp PRIVATE -Wall -Wextra -Werror -g1 -fno-omit-frame-pointer -fno-fast-math -ffp-contract=off)
endif()
add_library(recomp_rdp_gles2 STATIC backends/rdp_gles2.c backends/input.c include/rdp_gles2.h)
target_link_libraries(recomp_rdp_gles2 PUBLIC recomp_rdp SDL2::SDL2)
target_compile_definitions(recomp_rdp_gles2 PRIVATE SDL_MAIN_HANDLED)
if(MSVC)
    target_compile_options(recomp_rdp_gles2 PRIVATE /W4 /fp:strict)
else()
    target_compile_options(recomp_rdp_gles2 PRIVATE -Wall -Wextra -Werror -g1 -fno-omit-frame-pointer -fno-fast-math -ffp-contract=off)
endif()
option(RECOMP_BUILD_AUDIO "Build native ABI1/Zelda NEAD synthesis and SDL playback" ON)
if(RECOMP_BUILD_AUDIO)
    find_package(SDL2 REQUIRED)
    add_library(recomp_audio STATIC backends/audio.c third_party/rsp-hle/alist.c
        third_party/rsp-hle/alist_audio.c third_party/rsp-hle/alist_nead.c
        third_party/rsp-hle/alist_naudio.c third_party/rsp-hle/mp3.c
        third_party/rsp-hle/audio.c third_party/rsp-hle/musyx.c)
    target_include_directories(recomp_audio PRIVATE third_party/rsp-hle)
    file(GLOB RECOMP_RSP_HEADERS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/third_party/rsp-hle/*.h")
    target_sources(recomp_audio PRIVATE include/recomp_audio.h ${RECOMP_RSP_HEADERS})
    target_link_libraries(recomp_audio PUBLIC recomp_runtime recomp_rsp_memory SDL2::SDL2)
    if(MSVC)
        target_compile_options(recomp_audio PRIVATE /W4 /fp:strict)
    else()
        target_compile_options(recomp_audio PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
        target_compile_options(recomp_audio PRIVATE -g1 -fno-omit-frame-pointer)
    endif()
endif()

add_library(recomp_renderer STATIC backends/gles2.c)
target_link_libraries(recomp_renderer PUBLIC recomp_runtime recomp_rsp recomp_rdp_gles2 SDL2::SDL2)
if(MSVC)
    target_compile_options(recomp_renderer PRIVATE /W4)
else()
    target_compile_options(recomp_renderer PRIVATE -Wall -Wextra -Werror -g1 -fno-omit-frame-pointer)
endif()

foreach(target recomp_runtime recomp_rsp_memory recomp_jpeg recomp_rsp recomp_rdp recomp_rdp_gles2 recomp_audio recomp_renderer )
    if(TARGET "${target}")
        recomp_source_groups("${target}")
        recomp_optimize_release_target("${target}")
    endif()
endforeach()

if(NOT RECOMP_GAME STREQUAL "none")
    if(NOT TARGET recomp_renderer OR NOT RECOMP_BUILD_AUDIO)
        message(FATAL_ERROR "A playable game requires a renderer and RECOMP_BUILD_AUDIO")
    endif()
    add_subdirectory("games/${RECOMP_GAME}")
endif()
