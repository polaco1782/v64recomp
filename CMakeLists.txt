cmake_minimum_required(VERSION 3.20)
# Select defaults before project() enables C; preserve explicit compiler/toolchain choices.
if(NOT CMAKE_TOOLCHAIN_FILE AND "$ENV{CMAKE_TOOLCHAIN_FILE}" STREQUAL ""
        AND NOT DEFINED CMAKE_C_COMPILER AND "$ENV{CC}" STREQUAL "")
    find_program(CMAKE_C_COMPILER NAMES clang gcc REQUIRED)
endif()
project(v64recomp LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
configure_file(LICENSE "${CMAKE_BINARY_DIR}/bin/LICENSE" COPYONLY)
configure_file(THIRD_PARTY_NOTICES.md "${CMAKE_BINARY_DIR}/bin/THIRD_PARTY_NOTICES.md" COPYONLY)
configure_file(licenses/GLideN64.txt "${CMAKE_BINARY_DIR}/bin/licenses/GLideN64.txt" COPYONLY)
configure_file(licenses/Mupen64Plus-rsp-hle.txt "${CMAKE_BINARY_DIR}/bin/licenses/Mupen64Plus-rsp-hle.txt" COPYONLY)

# One native game per build tree. Use a separate directory for each mode.
set(RECOMP_GAME "sm64-us" CACHE STRING "Game to build, or none for shared libraries")
set(RECOMP_GAMES sm64-us goldeneye-us shadowman-us quake-us duke-us nightmare-us oot-us-v12 perfectdark-us-v11 conker-us)
set_property(CACHE RECOMP_GAME PROPERTY STRINGS none ${RECOMP_GAMES})
if(NOT RECOMP_GAME STREQUAL "none" AND NOT RECOMP_GAME IN_LIST RECOMP_GAMES)
    message(FATAL_ERROR "Unknown RECOMP_GAME=${RECOMP_GAME}. Choose: ${RECOMP_GAMES};none")
endif()
if(CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Use a single-configuration generator such as Ninja; release/debug select different application sources.")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Release or Debug" FORCE)
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|Debug)$")
    message(FATAL_ERROR "CMAKE_BUILD_TYPE must be Release or Debug")
endif()
set(RECOMP_RUN_ARGS "" CACHE STRING "Extra executable arguments for the run target (semicolon separated)")
set(RECOMP_RENDER_DEFAULT OFF)
if(NOT RECOMP_GAME STREQUAL "none")
    set(RECOMP_RENDER_DEFAULT ON)
endif()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(recomp_runtime STATIC runtime/memory.c runtime/dispatch.c runtime/fpu.c runtime/scheduler.c runtime/worker.c
    runtime/os.c runtime/bootstrap.c runtime/time.c runtime/platform.c runtime/tasks.c runtime/io.c runtime/video.c)
target_link_libraries(recomp_runtime PUBLIC OpenSSL::Crypto Threads::Threads m ${CMAKE_DL_LIBS})
target_include_directories(recomp_runtime PUBLIC include)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(recomp_runtime PUBLIC RECOMP_DEBUG)
    target_sources(recomp_runtime PRIVATE runtime/debug.c)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(recomp_runtime PRIVATE -Wall -Wextra -Werror -fno-fast-math -frounding-math -ffp-contract=off)
endif()
add_library(recomp_rsp_memory STATIC third_party/rsp-hle/memory.c)
target_compile_options(recomp_rsp_memory PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
add_library(recomp_jpeg STATIC backends/jpeg.c third_party/rsp-hle/jpeg.c)
target_include_directories(recomp_jpeg PRIVATE third_party/rsp-hle)
target_link_libraries(recomp_jpeg PUBLIC recomp_runtime recomp_rsp_memory)
target_compile_options(recomp_jpeg PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing -fno-fast-math -ffp-contract=off)
set_property(SOURCE third_party/rsp-hle/jpeg.c APPEND PROPERTY COMPILE_DEFINITIONS
    rsp_break=recomp_jpeg_break HleWarnMessage=recomp_jpeg_warn HleVerboseMessage=recomp_jpeg_verbose)
option(RECOMP_BUILD_AUDIO "Build native ABI1/Zelda NEAD synthesis and SDL playback" ON)
if(RECOMP_BUILD_AUDIO)
    find_package(SDL2 REQUIRED)
    add_library(recomp_audio STATIC backends/audio.c third_party/rsp-hle/alist.c
        third_party/rsp-hle/alist_audio.c third_party/rsp-hle/alist_nead.c
        third_party/rsp-hle/alist_naudio.c third_party/rsp-hle/mp3.c
        third_party/rsp-hle/audio.c)
    target_include_directories(recomp_audio PRIVATE third_party/rsp-hle)
    target_link_libraries(recomp_audio PUBLIC recomp_runtime recomp_rsp_memory SDL2::SDL2)
    target_compile_options(recomp_audio PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
endif()

option(RECOMP_BUILD_RT64 "Build the standalone RT64 graphics adapter" ${RECOMP_RENDER_DEFAULT})
if(RECOMP_BUILD_RT64)
    if(NOT CMAKE_TOOLCHAIN_FILE AND "$ENV{CMAKE_TOOLCHAIN_FILE}" STREQUAL ""
            AND NOT DEFINED CMAKE_CXX_COMPILER AND "$ENV{CXX}" STREQUAL "")
        find_program(CMAKE_CXX_COMPILER NAMES clang++ g++ REQUIRED)
    endif()
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(RECOMP_RT64_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/rt64" CACHE PATH "Existing RT64 source checkout")
    set(RT64_STATIC ON CACHE BOOL "" FORCE)
    set(RT64_SDL_WINDOW_VULKAN OFF CACHE BOOL "" FORCE)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    add_subdirectory("${RECOMP_RT64_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/rt64" EXCLUDE_FROM_ALL)
    find_package(SDL2 REQUIRED)
    add_library(recomp_renderer STATIC backends/rt64.cpp backends/textures.cpp
        texture_scan/scanner.cpp texture_scan/capture.cpp texture_scan/pack.cpp texture_scan/source.cpp)
    target_link_libraries(recomp_renderer PUBLIC recomp_runtime texture_hashing SDL2::SDL2)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_compile_definitions(recomp_renderer PRIVATE RECOMP_DEBUG)
    endif()
    target_compile_definitions(recomp_renderer PRIVATE HLSL_CPU FFX_GCC IMGUI_IMPL_VULKAN_NO_PROTOTYPES)
    target_compile_options(recomp_renderer PRIVATE -Wall -Wextra -Werror)
    set(RECOMP_RT64_INCLUDES
        "${RECOMP_RT64_SOURCE}/src" "${RECOMP_RT64_SOURCE}/src/contrib"
        "${RECOMP_RT64_SOURCE}/src/contrib/plume" "${RECOMP_RT64_SOURCE}/src/contrib/imgui"
        "${RECOMP_RT64_SOURCE}/src/contrib/hlslpp/include"
        "${RECOMP_RT64_SOURCE}/src/contrib/mupen64plus-core/src/api")
    target_include_directories(recomp_renderer SYSTEM PRIVATE ${RECOMP_RT64_INCLUDES})
endif()

if(NOT RECOMP_GAME STREQUAL "none")
    if(NOT RECOMP_BUILD_RT64 OR NOT RECOMP_BUILD_AUDIO)
        message(FATAL_ERROR "A playable game requires RECOMP_BUILD_RT64 and RECOMP_BUILD_AUDIO")
    endif()
    add_subdirectory("games/${RECOMP_GAME}")
endif()
