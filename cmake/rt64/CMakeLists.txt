cmake_minimum_required(VERSION 3.20)
project(v64recomp_rt64 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RT64_STATIC ON CACHE BOOL "" FORCE)
set(RT64_SDL_WINDOW_VULKAN OFF CACHE BOOL "" FORCE)
if(MSVC)
    option(RECOMP_MSVC_DEBUG_ITERATORS "Use MSVC's default STL iterator diagnostics" OFF)
    if(NOT RECOMP_MSVC_DEBUG_ITERATORS)
        add_compile_definitions("$<$<COMPILE_LANGUAGE:CXX>:_ITERATOR_DEBUG_LEVEL=0>")
    endif()
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
    add_compile_options(/utf-8)
endif()
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "${CMAKE_BUILD_TYPE}" CACHE STRING "Renderer mode" FORCE)
endif()

if(NOT EXISTS "${RECOMP_RT64_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Populate RECOMP_RT64_SOURCE before building RT64")
endif()
if(WIN32)
    find_package(SDL2 CONFIG REQUIRED)
endif()
add_subdirectory("${RECOMP_RT64_SOURCE}" rt64)
if(MSVC AND NOT RECOMP_MSVC_DEBUG_ITERATORS)
    # Export the STL ABI requirement to every consumer of the shared renderer.
    target_compile_definitions(rt64 PUBLIC _ITERATOR_DEBUG_LEVEL=0)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/StableProjection.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/PrimitiveDepth.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/LookAtMatching.cmake")

# Keep optimized native renderer stacks usable by the built-in CPU sampler.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(rt64 PRIVATE -g1 -fno-omit-frame-pointer)
    target_compile_options(plume PRIVATE -g1 -fno-omit-frame-pointer)
endif()

# Export the entire static link graph, including platform libraries and include
# requirements. Game builds import these targets rather than configuring RT64.
export(TARGETS rt64 re-spirv nfd libzstd_static plume
    NAMESPACE RecompRT64:: FILE "${CMAKE_BINARY_DIR}/rt64-targets.cmake")
